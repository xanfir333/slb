function cleanText(value) {

    if (value === null || value === undefined) {
        return "";
    }

    return String(value)
        .replace(/\u00A0/g, " ")
        .replace(/\r/g, "")
        .replace(/[ \t]+/g, " ")
        .trim();
}


function escapeHTML(value) {

    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}


function normalize(value) {

    return cleanText(value)
        .toLowerCase()
        .replace(/[^a-z0-9]/g, "");
}


function normalizeCountry(value) {

    return cleanText(value)
        .replace(/\s+/g, " ")
        .trim();
}


/* =========================================================
   DATE FUNCTIONS
========================================================= */

function formatDate(value) {

    if (value === null || value === undefined || value === "") {
        return "";
    }

    if (value instanceof Date && !isNaN(value)) {

        const d = String(value.getDate()).padStart(2, "0");
        const m = String(value.getMonth() + 1).padStart(2, "0");
        const y = value.getFullYear();

        return `${d}-${m}-${y}`;
    }

    let text = cleanText(value);

    /*
        Excel serial date
    */

    if (/^\d+(\.\d+)?$/.test(text)) {

        const number = Number(text);

        if (number > 20000 && number < 60000) {

            const date =
                new Date(
                    Math.round(
                        (number - 25569) * 86400 * 1000
                    )
                );

            if (!isNaN(date)) {

                return formatDate(date);
            }
        }
    }

    /*
        DD-MMM-YYYY
        25-Sep-2026
    */

    const monthMap = {
        jan: 0,
        feb: 1,
        mar: 2,
        apr: 3,
        may: 4,
        jun: 5,
        jul: 6,
        aug: 7,
        sep: 8,
        oct: 9,
        nov: 10,
        dec: 11
    };

    let match =
        text.match(
            /^(\d{1,2})[-\/ ]([A-Za-z]{3,})[-\/ ](\d{4})$/
        );

    if (match) {

        const day = Number(match[1]);
        const monthName = match[2].substring(0,3).toLowerCase();
        const year = Number(match[3]);

        if (monthMap[monthName] !== undefined) {

            return `${String(day).padStart(2,"0")}-${String(monthMap[monthName] + 1).padStart(2,"0")}-${year}`;
        }
    }

    /*
        DD-MM-YYYY / DD/MM/YYYY
    */

    match =
        text.match(
            /^(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})$/
        );

    if (match) {

        return `${String(match[1]).padStart(2,"0")}-${String(match[2]).padStart(2,"0")}-${match[3]}`;
    }

    return text;
}


/* =========================================================
   PDF EXTRACTION
========================================================= */

async function extractPDFText(file) {

    const buffer =
        await file.arrayBuffer();

    const pdf =
        await pdfjsLib.getDocument({
            data: buffer
        }).promise;

    let pages = [];

    for (let pageNumber = 1;
         pageNumber <= pdf.numPages;
         pageNumber++) {

        const page =
            await pdf.getPage(pageNumber);

        const content =
            await page.getTextContent();

        const items =
            content.items
                .map(item => cleanText(item.str))
                .filter(Boolean);

        pages.push(items.join("\n"));
    }

    return pages.join("\n");
}

function findMRN(pdfText) {

    const match =
        pdfText.match(
            /\bMRN\s+([A-Z0-9]{15,25})\b/i
        );

    return match
        ? cleanText(match[1])
        : "";
}


function findTransportNumber(pdfText) {

    /*
        TR26GBNL0001368
        TR + year + country/location + number
    */

    const match =
        pdfText.match(
            /\bTR\d{2}[A-Z0-9]{8,30}\b/i
        );

    return match
        ? cleanText(match[0])
        : "";
}


function findDeclarationDate(pdfText) {

    const match =
        pdfText.match(
            /Datum:\s*(\d{1,2}[-\/]\d{1,2}[-\/]\d{4})/i
        );

    if (match) {

        return formatDate(match[1]);
    }

    return "";
}


function findDeclarant(pdfText) {

    /*
        Real document structure:

        Declarant Vertegenwoordiger
        ...
        Naam: SCHLUMBERGER OFFSHORE SERVICES LTD
        ...
        Naam: PETERSON DEN HELDER BV
    */

    const match =
        pdfText.match(
            /Declarant\s+Vertegenwoordiger[\s\S]{0,500}?Naam:\s*([^\n]+?)\s*(?:Adres:|Postcode:)/i
        );

    if (match) {

        return cleanText(match[1]);
    }

    return "";
}


function findImporter(pdfText) {

    const match =
        pdfText.match(
            /Importeur\s+Exporteur[\s\S]{0,500}?Naam:\s*([^\n]+?)\s*(?:Adres:|Postcode:)/i
        );

    if (match) {

        return cleanText(match[1]);
    }

    return "";
}


function findPermanentTemporary(pdfText) {

    /*
        In this document:
        Regeling(en): 40 - 00

        Invoice:
        Export Type: Permanent Export

        Therefore the final tracker value is Permanent.
    */

    if (/Permanent\s+Export/i.test(pdfText)) {
        return "Permanent";
    }

    if (/\b40\s*-\s*00\b/.test(pdfText)) {
        return "Permanent";
    }

    if (/Temporary/i.test(pdfText)) {
        return "Temporary";
    }

    return "";
}


/* =========================================================
   PDF ARTICLE EXTRACTION
========================================================= */

function extractCustomsArticles(pdfText) {

    const lines =
        pdfText
            .split("\n")
            .map(cleanText)
            .filter(Boolean);

    const articles = [];

    let current = null;

    for (let i = 0; i < lines.length; i++) {

        const line = lines[i];

        let articleMatch =
            line.match(/^Artikel:\s*(\d+)$/i);

        if (articleMatch) {

            if (current) {
                articles.push(current);
            }

            current = {
                article: articleMatch[1],
                customsCode: "",
                country: "",
                description: "",
                quantity: null,
                gross: "",
                net: "",
                value: "",
                statisticalValue: ""
            };

            continue;
        }

        if (!current) {
            continue;
        }

        let match;

        match =
            line.match(
                /^Goederencode:\s*([0-9A-Z ]+?)\s+Land van oorsprong:\s*(.+)$/i
            );

        if (match) {

            current.customsCode =
                cleanText(match[1]);

            current.country =
                cleanText(match[2]);

            continue;
        }

        match =
            line.match(
                /^Omschrijving:\s*(.+)$/i
            );

        if (match) {

            current.description =
                cleanText(match[1]);

            const qty =
                match[1].match(
                    /^(\d+)\s+PCS\b/i
                );

            if (qty) {
                current.quantity =
                    Number(qty[1]);
            }

            continue;
        }

        match =
            line.match(
                /^Bruto:\s*([\d.,]+)\s*kg/i
            );

        if (match) {

            current.gross =
                cleanText(match[1]);

            continue;
        }

        match =
            line.match(
                /^Netto:\s*([\d.,]+)\s*kg/i
            );

        if (match) {

            current.net =
                cleanText(match[1]);

            continue;
        }

        match =
            line.match(
                /^Waarde:\s*([\d.,]+)\s*EUR/i
            );

        if (match) {

            current.value =
                cleanText(match[1]);

            continue;
        }

        match =
            line.match(
                /^Statistische waarde:\s*([\d.,]+)\s*EUR/i
            );

        if (match) {

            current.statisticalValue =
                cleanText(match[1]);

            continue;
        }
    }

    if (current) {
        articles.push(current);
    }

    return articles;
}

    if (value === null || value === undefined) {

    }

    return cleanText(value);
}

/* =========================================================
   EXACT INVOICE STRUCTURE
========================================================= */

function extractExactInvoiceData(rows) {

    /*
        Real invoice structure:

        C = Line Item Number
        D = PO Number - SO Number
        F = Part Number
        H = Description & Identifiers
        K = UN Number
        M = Customs Data
        O = Quantity
        Q = Unit of Measure
        T = Unit Value
        U = Line Value

        Metadata:

        T8  = Transport Number
        T9  = Invoice Number
        T11 = Date
        T16 = Export Type

        C161 = Notes / Comments
        Receiving Cost Center appears there.
    */

    const result = {

        transportNumber: "",
        invoiceNumber: "",
        invoiceDate: "",
        exportType: "",
        productLine: "",
        products: []
    };


    /*
        Metadata
    */

    result.transportNumber =
        cellText(rows[7]?.[19]);

    result.invoiceNumber =
        cellText(rows[8]?.[19]);

    result.invoiceDate =
        formatDate(rows[10]?.[19]);

    result.exportType =
        cellText(rows[15]?.[19]);


    /*
        Receiving Cost Center

        Example:

        Receiving Cost Center:
        NL100123 TDBM
    */

    for (let r = 0; r < rows.length; r++) {

        const joined =
            rows[r]
                .map(cellText)
                .join(" ");

        if (/Receiving Cost Center/i.test(joined)) {

            let found =
                joined.match(
                    /Receiving Cost Center:\s*([A-Z0-9-]+(?:\s+[A-Z0-9]+)?)/i
                );

            if (found) {

                result.productLine =
                    cleanText(found[1]);
            }

            /*
                If the value is on the next row/cell
            */

            if (!result.productLine) {

                for (let rr = r; rr <= Math.min(r + 3, rows.length - 1); rr++) {

                    for (let cc = 0; cc < rows[rr].length; cc++) {

                        const value =
                            cellText(rows[rr][cc]);

                        if (
                            /^NL\d{5}\b/i.test(value)
                        ) {

                            result.productLine =
                                cleanText(value);

                            break;
                        }
                    }

                    if (result.productLine) {
                        break;
                    }
                }
            }
        }
    }


    /*
        Exact invoice line positions.

        Start at row 31 in Excel = index 30 in JS.
        Each item occupies approximately 8 rows.
    */

    for (let r = 30; r < rows.length; r++) {

        const lineNumber =
            cellText(rows[r]?.[2]);

        /*
            A valid invoice line begins with
            an integer in column C.
        */

        if (!/^\d+$/.test(lineNumber)) {
            continue;
        }

        const partNumber =
            cellText(rows[r]?.[5]);

        const rawDescription =
            cellText(rows[r]?.[7]);

        const customsData =
            cellText(rows[r]?.[12]);

        const quantity =
            cellText(rows[r]?.[14]);

        const unit =
            cellText(rows[r]?.[16]);

        const unitValue =
            cellText(rows[r]?.[19]);

        const lineValue =
            cellText(rows[r]?.[20]);

        /*
            Extract serial number.
        */

        let serialNumber = "";

        const serialMatch =
            rawDescription.match(
                /S\/N:\s*([^\n]*)/i
            );

        if (serialMatch) {

            serialNumber =
                cleanText(serialMatch[1]);
        }

        /*
            Empty S/N means N/A.
        */

        if (!serialNumber) {
            serialNumber = "N/A";
        }


        /*
            First actual description line.
        */

        const descriptionLines =
            rawDescription
                .split(/\n/)
                .map(cleanText)
                .filter(Boolean);

        let itemDescription =
            descriptionLines.length
                ? descriptionLines[0]
                : "";


        /*
            Extract Import HTC Code.
        */

        let importHTC = "";

        let match =
            customsData.match(
                /Import HTC Code:\s*([A-Z0-9]+)/i
            );

        if (match) {
            importHTC =
                cleanText(match[1]);
        }


        /*
            Extract Export HTC Code.
        */

        let exportHTC = "";

        match =
            customsData.match(
                /Export HTC Code:\s*([A-Z0-9]+)/i
            );

        if (match) {
            exportHTC =
                cleanText(match[1]);
        }


        /*
            Extract Country of Origin.
        */

        let countryOfOrigin = "";

        match =
            customsData.match(
                /Country of Origin:\s*([^\n]+)/i
            );

        if (match) {
            countryOfOrigin =
                cleanText(match[1]);
        }


        /*
            Extract ECCN.
        */

        let eccn = "";

        match =
            customsData.match(
                /ECCN:\s*([^\n]+)/i
            );

        if (match) {
            eccn =
                cleanText(match[1]);
        }


        /*
            Extract FCG.
        */

        let fcg = "";

        match =
            customsData.match(
                /FCG[^:]*-\s*([^\n]+)/i
            );

        if (match) {
            fcg =
                cleanText(match[1]);
        }


        /*
            Extract PO/SO.
        */

        const poSo =
            cellText(rows[r]?.[3]);


        result.products.push({

            number: lineNumber,

            poSo: poSo,

            partNumber: partNumber,

            serialNumber: serialNumber,

            description: itemDescription,

            fullDescription: rawDescription,

            quantity: quantity,

            unit: unit,

            unitValue: unitValue,

            lineValue: lineValue,

            importHTC: importHTC,

            exportHTC: exportHTC,

            countryOfOrigin: countryOfOrigin,

            eccn: eccn,

            fcg: fcg

        });
    }


    return result;
}


/* =========================================================
   GENERIC FALLBACK FOR OTHER EXCEL FILES
========================================================= */

function findHeaderRow(rows) {

    const aliases = {

        number: [
            "line item number",
            "line number",
            "line no",
            "line",
            "item number",
            "item no",
            "no"
        ],

        partNumber: [
            "part number",
            "part no",
            "part",
            "material number",
            "material no"
        ],

        description: [
            "description & identifiers",
            "item description",
            "description",
            "product description"
        ],

        customsData: [
            "customs data"
        ],

        quantity: [
            "quantity",
            "qty"
        ]
    };


    let bestRow = -1;
    let bestScore = 0;

    for (
        let r = 0;
        r < Math.min(rows.length, 80);
        r++
    ) {

        let score = 0;

        const row =
            rows[r].map(cellText);

        row.forEach(cell => {

            const n =
                normalize(cell);

            for (const key in aliases) {

                if (
                    aliases[key].some(
                        alias =>
                            normalize(alias) === n
                    )
                ) {

                    score++;
                }
            }
        });

        if (score > bestScore) {

            bestScore = score;
            bestRow = r;
        }
    }

    return bestRow;
}


function extractGenericInvoiceData(rows) {

    const result = {

        transportNumber: "",
        invoiceNumber: "",
        invoiceDate: "",
        exportType: "",
        productLine: "",
        products: []
    };


    /*
        Search metadata throughout workbook.
    */

    for (let r = 0; r < rows.length; r++) {

        for (let c = 0; c < rows[r].length; c++) {

            const label =
                normalize(rows[r][c]);

            const next =
                cellText(rows[r][c + 1]);

            if (label === "transportnumber") {

                result.transportNumber = next;
            }

            if (label === "invoicenumber") {

                result.invoiceNumber = next;
            }

            if (label === "date") {

                const possibleDate =
                    formatDate(next);

                if (possibleDate) {
                    result.invoiceDate = possibleDate;
                }
            }

            if (
                label === "exporttype" ||
                label === "typeofexport"
            ) {

                result.exportType = next;
            }
        }
    }


    const headerRow =
        findHeaderRow(rows);

    if (headerRow < 0) {
        return result;
    }


    const headers =
        rows[headerRow].map(cellText);


    function findColumn(names) {

        for (let c = 0; c < headers.length; c++) {

            const n =
                normalize(headers[c]);

            if (
                names.some(
                    name =>
                        normalize(name) === n
                )
            ) {
                return c;
            }
        }

        return -1;
    }


    const colNumber =
        findColumn([
            "Line Item Number",
            "Line Number",
            "Line",
            "No"
        ]);

    const colPart =
        findColumn([
            "Part Number",
            "Part No",
            "Part"
        ]);

    const colDescription =
        findColumn([
            "Description & Identifiers",
            "Description",
            "Item Description"
        ]);

    const colCustoms =
        findColumn([
            "Customs Data"
        ]);

    const colQuantity =
        findColumn([
            "Quantity",
            "Qty"
        ]);


    for (
        let r = headerRow + 1;
        r < rows.length;
        r++
    ) {

        const row =
            rows[r];

        const number =
            colNumber >= 0
                ? cellText(row[colNumber])
                : "";

        if (
            !number ||
            !/^\d+$/.test(number)
        ) {
            continue;
        }

        const partNumber =
            colPart >= 0
                ? cellText(row[colPart])
                : "";

        const description =
            colDescription >= 0
                ? cellText(row[colDescription])
                : "";

        const customsData =
            colCustoms >= 0
                ? cellText(row[colCustoms])
                : "";

        const quantity =
            colQuantity >= 0
                ? cellText(row[colQuantity])
                : "";


        let serialNumber = "N/A";

        const serialMatch =
            description.match(
                /S\/N:\s*([^\n]*)/i
            );

        if (serialMatch) {

            const serial =
                cleanText(serialMatch[1]);

            if (serial) {
                serialNumber = serial;
            }
        }


        const descriptionLines =
            description
                .split(/\n/)
                .map(cleanText)
                .filter(Boolean);


        let itemDescription =
            descriptionLines.length
                ? descriptionLines[0]
                : "";


        let importHTC = "";

        const importMatch =
            customsData.match(
                /Import HTC Code:\s*([A-Z0-9]+)/i
            );

        if (importMatch) {

            importHTC =
                cleanText(importMatch[1]);
        }


        result.products.push({

            number,
            partNumber,
            serialNumber,
            description: itemDescription,
            quantity,
            importHTC

        });
    }


    return result;
}


/* =========================================================
   READ EXCEL
========================================================= */

async function extractExcelData(file) {

    const buffer =
        await file.arrayBuffer();

    const workbook =
        XLSX.read(
            buffer,
            {
                type: "array",
                cellDates: true
            }
        );


    /*
        Try all sheets.
    */

    let bestResult = null;


    for (const sheetName of workbook.SheetNames) {

        const sheet =
            workbook.Sheets[sheetName];

        const rows =
            getSheetRows(sheet);


        /*
            First attempt:
            exact known invoice structure.
        */

        let result =
            extractExactInvoiceData(rows);


        /*
            If exact structure didn't find products,
            use generic extraction.
        */

        if (!result.products.length) {

            result =
                extractGenericInvoiceData(rows);
        }


        /*
            Choose the sheet with the most
            detected invoice lines.
        */

        if (
            !bestResult ||
            result.products.length >
            bestResult.products.length
        ) {

            bestResult = result;
        }
    }


    return bestResult || {
        transportNumber: "",
        invoiceNumber: "",
        invoiceDate: "",
        exportType: "",
        productLine: "",
        products: []
    };
}


/* =========================================================
   CUSTOMS / INVOICE CROSS-CHECK
========================================================= */

function normalizeHTC(code) {

    return cleanText(code)
        .toUpperCase()
        .replace(/[^0-9]/g, "");
}


function htcCompatible(invoiceCode, customsCode) {

    const a =
        normalizeHTC(invoiceCode);

    const b =
        normalizeHTC(customsCode);

    if (!a || !b) {
        return false;
    }

    /*
        Exact match
    */

    if (a === b) {
        return true;
    }

    /*
        Import code can be shorter than
        the declaration's 10-digit CN code.

        Example:
        90309000
        9030900000
    */

    if (
        b.startsWith(a) ||
        a.startsWith(b)
    ) {
        return true;
    }

    /*
        Handle the specific formatting
        difference seen for 860900090
        vs 8609009000.

        Compare first six digits.
    */

    if (
        a.length >= 6 &&
        b.length >= 6 &&
        a.substring(0,6) ===
        b.substring(0,6)
    ) {

        return true;
    }

    return false;
}


function crossCheckInvoiceAndCustoms(
    invoiceProducts,
    customsArticles
) {

    const mappings = [];

    for (const product of invoiceProducts) {

        const candidates =
            customsArticles.filter(
                article =>
                    htcCompatible(
                        product.importHTC,
                        article.customsCode
                    )
            );


        if (!candidates.length) {

            mappings.push({

                product,
                customsArticle: null,
                status: "NO MATCH"

            });

            continue;
        }


        /*
            If multiple customs articles have
            the same code, prefer one where
            country matches.
        */

        let best =
            candidates.find(
                article =>
                    normalizeCountry(
                        article.country
                    ).toLowerCase()
                    .includes(
                        normalizeCountry(
                            product.countryOfOrigin || ""
                        ).toLowerCase()
                    )
            );


        if (!best) {
            best = candidates[0];
        }


        mappings.push({

            product,

            customsArticle: best,

            status: "MATCH"

        });
    }


    return mappings;
}


/* =========================================================
   GENERATE TRACKER
========================================================= */

async function generateTracker() {

    const pdfFile =
        document.getElementById("pdfFile").files[0];

    const excelFile =
        document.getElementById("excelFile").files[0];


    if (!pdfFile) {

        alert(
            "Please select the Customs Declaration PDF."
        );

        return;
    }


    if (!excelFile) {

        alert(
            "Please select the Commercial Invoice Excel."
        );

        return;
    }


    const status =
        document.getElementById("status");

    const validation =
        document.getElementById("validation");


    try {

        status.innerText =
            "Reading Customs Declaration...";


        /*
            Read PDF completely.
        */

        const pdfText =
            await extractPDFText(pdfFile);


        status.innerText =
            "Reading Commercial Invoice...";


        /*
            Read Excel completely.
        */

        const invoice =
            await extractExcelData(excelFile);


        /*
            Extract Customs Declaration.
        */

        const customs = {

            mrn:
                findMRN(pdfText),

            date:
                findDeclarationDate(pdfText),

            transportNumber:
                findTransportNumber(pdfText),

            declarant:
                findDeclarant(pdfText),

            importer:
                findImporter(pdfText),

            temporaryPermanent:
                findPermanentTemporary(pdfText),

            articles:
                extractCustomsArticles(pdfText)

        };


        /*
            Invoice transport number has priority
            if PDF does not contain one.
        */

        const transportNumber =
            customs.transportNumber ||
            invoice.transportNumber;


        /*
            Product Line:
            receiving cost center from invoice.
        */

        const productLine =
            invoice.productLine;


        /*
            Cross-check.
        */

        const mappings =
            crossCheckInvoiceAndCustoms(
                invoice.products,
                customs.articles
            );


        /*
            Build table.
        */

        const tbody =
            document.getElementById("tbody");

        tbody.innerHTML = "";


        invoice.products.forEach(
            (product, index) => {

                const row =
                    document.createElement("tr");


                row.innerHTML = `

<td>
    ${escapeHTML(transportNumber)}
</td>

<td>
    ${escapeHTML(customs.mrn)}
</td>

<td>
    ${escapeHTML(customs.date)}
</td>

<td>
    ${escapeHTML(customs.declarant)}
</td>

<td>
    ${escapeHTML(customs.importer)}
</td>

<td>
    ${escapeHTML(customs.temporaryPermanent)}
</td>

<td>
    ${escapeHTML(productLine)}
</td>

<td>
    ${escapeHTML(product.number || String(index + 1))}
</td>

<td>
    ${escapeHTML(invoice.invoiceDate)}
</td>

<td>
    ${escapeHTML(product.partNumber)}
</td>

<td>
    ${escapeHTML(product.serialNumber || "N/A")}
</td>

<td>
    ${escapeHTML(product.description)}
</td>

<td contenteditable="true" class="manual"></td>

<td contenteditable="true" class="manual"></td>

<td contenteditable="true" class="manual"></td>

<td contenteditable="true" class="manual"></td>

                `;


                tbody.appendChild(row);

            }
        );


        /*
            Validation report.
        */

        const matched =
            mappings.filter(
                x => x.status === "MATCH"
            ).length;


        const unmatched =
            mappings.length - matched;


        validation.style.display =
            "block";


        if (
            invoice.products.length > 0 &&
            unmatched === 0
        ) {

            validation.className =
                "validation-ok";

        } else if (
            matched > 0
        ) {

            validation.className =
                "validation-warning";

        } else {

            validation.className =
                "validation-error";
        }


        /*
            Customs article summary.
        */

        const articleSummary =
            customs.articles
                .map(
                    a =>
                        `Article ${a.article}: HS ${a.customsCode}, ${a.quantity ?? "?"} PCS`
                )
                .join("\n");


        validation.innerText =

`DOCUMENT CHECK COMPLETE

Customs Declaration
MRN: ${customs.mrn || "Not found"}
Date: ${customs.date || "Not found"}
Transport: ${customs.transportNumber || "Not found"}
Declarant: ${customs.declarant || "Not found"}
Importer: ${customs.importer || "Not found"}
Regime: ${customs.temporaryPermanent || "Not found"}

Commercial Invoice
Invoice Number: ${invoice.invoiceNumber || "Not found"}
Invoice Date: ${invoice.invoiceDate || "Not found"}
Transport: ${invoice.transportNumber || "Not found"}
Product Line: ${productLine || "Not found"}
Invoice Lines Found: ${invoice.products.length}

Customs Articles Found: ${customs.articles.length}

Invoice / Customs Matches:
${matched} / ${invoice.products.length}

Unmatched Invoice Lines:
${unmatched}

Customs Articles:
${articleSummary || "None"}

Manual fields remain editable:
Transport Document / Guarantee Paid / Guarantee Claimed / Closed`;


        status.innerText =
            `Tracker generated successfully.

Invoice lines detected: ${invoice.products.length}
Customs articles detected: ${customs.articles.length}
Matched invoice lines: ${matched}
Unmatched invoice lines: ${unmatched}`;


        /*
            Console debugging information.
            Useful when a colleague has a different invoice.
        */

        console.log(
            "CUSTOMS DECLARATION:",
            customs
        );

        console.log(
            "COMMERCIAL INVOICE:",
            invoice
        );

        console.log(
            "CROSS-CHECK:",
            mappings
        );

    } catch (error) {

        console.error(error);

        status.innerText =
            "ERROR: " + error.message;

        validation.style.display =
            "block";

        validation.className =
            "validation-error";

        validation.innerText =
            "The documents could not be processed completely.\n\n" +
            error.message;
    }
}


/* =========================================================
   COPY TABLE
========================================================= */

async function copyTable() {

    const table =
        document.getElementById("trackerTable");


    let text = "";


    table.querySelectorAll("tr")
        .forEach(row => {

            const values = [];


            row.querySelectorAll("th, td")
                .forEach(cell => {

                    values.push(
                        cell.innerText
                            .replace(/\n/g, " ")
                            .trim()
                    );

                });


            text +=
                values.join("\t") +
                "\n";

        });


    try {

        await navigator.clipboard.writeText(text);

        alert(
            "Table copied successfully.\n\nOpen Excel and press CTRL + V."
        );

    } catch (error) {

        alert(
            "Could not copy automatically. Please select the table and copy it manually."
        );

    }
}
